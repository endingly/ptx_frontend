cmake_minimum_required(VERSION 3.28)
project(ptx_frontend LANGUAGES C CXX)

if(NOT PTX_SNAPSHOT_SOURCE_ROOT)
    message(FATAL_ERROR "Set PTX_SNAPSHOT_SOURCE_ROOT to the selected frontend source tree")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(BUILD_TESTING OFF CACHE BOOL "Only build the snapshot dependencies" FORCE)
list(PREPEND CMAKE_MODULE_PATH "${PTX_SNAPSHOT_SOURCE_ROOT}/cmake")
include(register_headers)
foreach(component IN ITEMS common base lexer cst syntax)
    add_subdirectory("${PTX_SNAPSHOT_SOURCE_ROOT}/submod/${component}" "${component}")
endforeach()
add_executable(parser_snapshot snapshot.cpp)
target_link_libraries(parser_snapshot PRIVATE ptx_frontend::syntax)
option(PTX_SNAPSHOT_GENERATED "Snapshot the private full generated parser entry" OFF)
if(PTX_SNAPSHOT_GENERATED)
    if(EXISTS "${PTX_SNAPSHOT_SOURCE_ROOT}/submod/cst/src/ptx_parser_driver.hpp")
        target_include_directories(parser_snapshot PRIVATE "${PTX_SNAPSHOT_SOURCE_ROOT}/submod/cst/src")
    elseif(EXISTS "${PTX_SNAPSHOT_SOURCE_ROOT}/submod/cst/bison/ptx_parser_driver.hpp")
        # Archived candidates retain their original private source layout.
        target_include_directories(parser_snapshot PRIVATE "${PTX_SNAPSHOT_SOURCE_ROOT}/submod/cst/bison")
    else()
        message(FATAL_ERROR "The selected source tree has no private generated parser driver")
    endif()
    target_compile_definitions(parser_snapshot PRIVATE PTX_SNAPSHOT_GENERATED=1)
endif()
